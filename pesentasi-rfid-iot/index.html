Praktikum 2: Push Button

Wiring & Konsep

Push Button + LED

Push Button:

• Pin 1 → ESP32 Pin 4
• Pin 1 → Resistor 10kΩ → 3.3V (pull-up)
• Pin 2 → GND

LED:

• ESP32 Pin 2 → Resistor 220Ω → LED Anode
• LED Cathode → GND

Konsep Pull-up Resistor

Tidak ditekan: Pin = HIGH (3.3V)
Ditekan: Pin = LOW (0V)
Pull-up: Mencegah floating input
Debouncing: Delay untuk stabilitas

Key Points

INPUT_PULLUP

Menggunakan resistor pull-up internal ESP32

!digitalRead(BUTTON_PIN)

Membalik logika karena pull-up (LOW = pressed)

delay(50)

Debouncing untuk menghindari pembacaan ganda

Code Program

// Praktikum 2: Push Button Input
#define BUTTON_PIN 4
#define LED_PIN 2

// Variabel untuk menyimpan status button
bool buttonState = false;
bool lastButtonState = false;
bool ledState = false;

void setup() {
  Serial.begin(115200);
  Serial.println("Praktikum 2: Push Button");
  
  // Set pin modes
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(LED_PIN, OUTPUT);
  
  // Matikan LED di awal
  digitalWrite(LED_PIN, LOW);
}

void loop() {
  // Baca status button (LOW = pressed)
  buttonState = !digitalRead(BUTTON_PIN);
  
  // Deteksi perubahan dari tidak ditekan ke ditekan
  if (buttonState && !lastButtonState) {
    // Toggle LED state
    ledState = !ledState;
    digitalWrite(LED_PIN, ledState);
    
    Serial.print("Button pressed! LED: ");
    Serial.println(ledState ? "ON" : "OFF");
  }
  
  // Simpan status button
  lastButtonState = buttonState;
  
  // Delay untuk debouncing
  delay(50);
}

// Fungsi tambahan untuk debouncing yang lebih baik
bool debounceButton(int pin, int delayTime = 50) {
  static bool lastState = HIGH;
  static unsigned long lastDebounceTime = 0;
  
  bool reading = digitalRead(pin);
  
  if (reading != lastState) {
    lastDebounceTime = millis();
  }
  
  if ((millis() - lastDebounceTime) > delayTime) {
    if (reading != buttonState) {
      buttonState = reading;
      return true; // State changed
    }
  }
  
  lastState = reading;
  return false; // No change
}

Komunikasi I2C

I2C Communication & RFID

Konsep I2C Protocol

Karakteristik I2C

SDA: Serial Data (bidirectional)
SCL: Serial Clock (master controlled)
Multi-device: Hingga 127 device
Addressing: 7-bit address unik
Speed: 100kHz - 1MHz

Pin I2C ESP32

SDA (Data) Pin 21
SCL (Clock) Pin 22

*Pin dapat dikonfigurasi ke pin lain jika diperlukan

Common I2C Addresses

Device Address (Hex) Address (Dec)
LCD 16x2 0x27 / 0x3F 39 / 63
PN532 NFC 0x24 36
OLED Display 0x3C / 0x3D 60 / 61
RTC DS3231 0x68 104
BME280 0x76 / 0x77 118 / 119

Keunggulan I2C

  • Hanya butuh 2 kabel (SDA, SCL)
  • Dapat menghubungkan banyak device
  • Built-in addressing system
  • Acknowledgment untuk error detection
  • Mudah di-debug dengan logic analyzer

I2C Scanner & Troubleshooting

I2C Scanner Code

i2c_scanner.ino 
#include <Wire.h>

void setup() {
  Wire.begin();
  Serial.begin(115200);
  Serial.println("I2C Scanner");
}

void loop() {
  byte error, address;
  int nDevices;
  
  Serial.println("Scanning...");
  nDevices = 0;
  
  for(address = 1; address < 127; address++) {
    Wire.beginTransmission(address);
    error = Wire.endTransmission();
    
    if (error == 0) {
      Serial.print("I2C device found at address 0x");
      if (address < 16) Serial.print("0");
      Serial.print(address, HEX);
      Serial.println(" !");
      nDevices++;
    }
  }
  
  if (nDevices == 0)
    Serial.println("No I2C devices found");
  else
    Serial.println("done");
    
  delay(5000);
}

Common Issues & Solutions

Problem: Device Not Found

  • • Check wiring connections
  • • Verify power supply (3.3V/5V)
  • • Check pull-up resistors (4.7kΩ)
  • • Confirm device address

Problem: Communication Error

  • • Check cable length (max 1m)
  • • Reduce I2C speed
  • • Check for noise interference
  • • Verify ground connections

Best Practices

  • • Use proper pull-up resistors
  • • Keep wires short and twisted
  • • Use common ground
  • • Test one device at a time

Pull-up Resistor Calculation

Formula: R = (Vcc - 0.4V) / (3mA × n_devices)

Typical values:

  • • 1-2 devices: 4.7kΩ
  • • 3-5 devices: 2.2kΩ
  • • 6+ devices: 1kΩ

Basic RFID Reader Code

rfid_basic_reader.ino 
#include <Wire.h>
#include <Adafruit_PN532.h>

// PN532 I2C pins
#define PN532_SDA 21
#define PN532_SCL 22

Adafruit_PN532 nfc(PN532_SDA, PN532_SCL);

void setup() {
  Serial.begin(115200);
  Serial.println("PN532 RFID Reader Test");
  
  nfc.begin();
  
  uint32_t versiondata = nfc.getFirmwareVersion();
  if (!versiondata) {
    Serial.print("Didn't find PN532 board");
    while (1); // halt
  }
  
  Serial.print("Found chip PN5"); 
  Serial.println((versiondata>>24) & 0xFF, HEX);
  Serial.print("Firmware ver. "); 
  Serial.print((versiondata>>16) & 0xFF, DEC);
  Serial.print('.'); 
  Serial.println((versiondata>>8) & 0xFF, DEC);
  
  // Configure board to read RFID tags
  nfc.SAMConfig();
  
  Serial.println("Waiting for an ISO14443A Card ...");
}

void loop() {
  uint8_t success;
  uint8_t uid[] = { 0, 0, 0, 0, 0, 0, 0 };
  uint8_t uidLength;
  
  success = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, 
                                    uid, &uidLength);
  
  if (success) {
    Serial.println("Found an ISO14443A card");
    Serial.print("  UID Length: ");
    Serial.print(uidLength, DEC);
    Serial.println(" bytes");
    Serial.print("  UID Value: ");
    
    for (uint8_t i=0; i < uidLength; i++) {
      Serial.print(" 0x");
      Serial.print(uid[i], HEX);
    }
    Serial.println("");
    
    delay(1000);
  }
}

Key Functions

nfc.begin() - Initialize PN532
nfc.getFirmwareVersion() - Check connection
nfc.SAMConfig() - Configure for RFID
nfc.readPassiveTargetID() - Read card UID

#include <WiFi.h>

const char* ssid = "YOUR_WIFI_SSID";

void setup() {
  Serial.begin(115200);
  
  // Connect to WiFi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println();
  Serial.println("WiFi connected!");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  // Check WiFi connection
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi disconnected. Reconnecting...");
    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) {
      delay(500);
      Serial.print(".");
    }
    Serial.println("Reconnected!");
  }
  
  delay(10000); // Check every 10 seconds
}

MQTT Implementation

mqtt_client.ino 
#include <WiFi.h>
#include <PubSubClient.h>

const char* mqtt_server = "broker.hivemq.com";
const int mqtt_port = 1883;
const char* mqtt_topic = "rfid/attendance";

WiFiClient espClient;
PubSubClient client(espClient);

void setup() {
  Serial.begin(115200);
  
  // Setup WiFi (previous code)
  setupWiFi();
  
  // Setup MQTT
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
}

void callback(char* topic, byte* payload, unsigned int length) {
  Serial.print("Message arrived [");
  Serial.print(topic);
  Serial.print("] ");
  
  String message;
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  Serial.println(message);
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection...");
    
    if (client.connect("ESP32Client")) {
      Serial.println("connected");
      client.subscribe("rfid/commands");
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" try again in 5 seconds");
      delay(5000);
    }
  }
}

void publishAttendance(String uid, String name) {
  String payload = "{\"uid\":\"" + uid + 
                   "\",\"name\":\"" + name + 
                   "\",\"timestamp\":" + String(millis()) + "}";
  
  client.publish(mqtt_topic, payload.c_str());
  Serial.println("Published: " + payload);
}

void loop() {
  if (!client.connected()) {
    reconnectMQTT();
  }
  client.loop();
  
  // Read RFID card
  uint8_t success;
  uint8_t uid[] = { 0, 0, 0, 0, 0, 0, 0 };
  uint8_t uidLength;
  
  success = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, 
                                    uid, &uidLength);
  
  if (success) {
    String cardUID = "";
    for (uint8_t i = 0; i < uidLength; i++) {
      if (i > 0) cardUID += ":";
      if (uid[i] < 0x10) cardUID += "0";
      cardUID += String(uid[i], HEX);
    }
    cardUID.toUpperCase();
    
    processAttendance(cardUID);
    delay(2000); // Prevent multiple reads
  }
}

void processAttendance(String cardUID) {
  String userName = findUser(cardUID);
  
  if (userName != "") {
    // Valid user found
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Welcome!");
    lcd.setCursor(0, 1);
    lcd.print(userName);
    
    // Success beep
    digitalWrite(BUZZER_PIN, HIGH);
    delay(200);
    digitalWrite(BUZZER_PIN, LOW);
    
    // Send to MQTT
    publishAttendance(cardUID, userName);
    
    Serial.println("Attendance recorded: " + userName);
  } else {
    // Unknown card
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Access Denied");
    lcd.setCursor(0, 1);
    lcd.print("Unknown Card");
    
    // Error beep
    for (int i = 0; i < 3; i++) {
      digitalWrite(BUZZER_PIN, HIGH);
      delay(100);
      digitalWrite(BUZZER_PIN, LOW);
      delay(100);
    }
    
    Serial.println("Unknown card: " + cardUID);
  }
  
  delay(2000);
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("System Ready");
  lcd.setCursor(0, 1);
  lcd.print("Scan your card");
}

String findUser(String uid) {
  for (int i = 0; i < sizeof(users)/sizeof(users[0]); i++) {
    if (users[i].uid == uid) {
      return users[i].name;
    }
  }
  return "";
}

void publishAttendance(String uid, String name) {
  String payload = "{";
  payload += "\"uid\":\"" + uid + "\",";
  payload += "\"name\":\"" + name + "\",";
  payload += "\"timestamp\":" + String(millis()) + ",";
  payload += "\"device\":\"ESP32_001\"";
  payload += "}";
  
  client.publish(mqtt_topic, payload.c_str());
}

const express = require('express');
const mqtt = require('mqtt');
const sqlite3 = require('sqlite3').verbose();
const path = require('path');

const app = express();
const port = 3000;

// Initialize SQLite database
const db = new sqlite3.Database('attendance.db');

// Create tables
db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS attendance (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    uid TEXT NOT NULL,
    name TEXT NOT NULL,
    timestamp INTEGER NOT NULL,
    device TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  )`);
  
  db.run(`CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    uid TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    email TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  )`);
});

// MQTT Client
const client = mqtt.connect('mqtt://broker.hivemq.com');

client.on('connect', () => {
  console.log('Connected to MQTT broker');
  client.subscribe('rfid/attendance');
});

client.on('message', (topic, message) => {
  if (topic === 'rfid/attendance') {
    try {
      const data = JSON.parse(message.toString());
      
      // Insert attendance record
      db.run(`INSERT INTO attendance 
              (uid, name, timestamp, device) 
              VALUES (?, ?, ?, ?)`,
        [data.uid, data.name, data.timestamp, data.device],
        function(err) {
          if (err) {
            console.error('Database error:', err);
          } else {
            console.log('Attendance recorded:', data.name);
          }
        }
      );
    } catch (error) {
      console.error('JSON parse error:', error);
    }
  }
});

// Middleware
app.use(express.json());
app.use(express.static('public'));

// API Routes
app.get('/api/attendance', (req, res) => {
  db.all(`SELECT * FROM attendance 
          ORDER BY created_at DESC 
          LIMIT 100`, (err, rows) => {
    if (err) {
      res.status(500).json({ error: err.message });
    } else {
      res.json(rows);
    }
  });
});

app.listen(port, () => {
  console.log(`Server running at http://localhost:${port}`);
});
